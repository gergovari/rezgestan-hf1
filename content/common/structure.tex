% Requires: \usepackage{tikz}, \usepackage{amssymb}

% --------------------------------------------------
% 1. PARAMETERS & SCALE
% --------------------------------------------------
\newcommand{\mul}{15}
\newcommand{\beamwidth}{5pt}
\newcommand{\beamcolor}{gray!80!black}

\newcommand{\lthree}{0.13}
\newcommand{\lfour}{0.33}
\newcommand{\lone}{0.18}
\newcommand{\ltwo}{0.1}

% --------------------------------------------------
% 2. COORDINATES
% --------------------------------------------------
\newcommand{\coords}{
    \pgfmathsetmacro{\Lthree}{\lthree * \mul}
    \pgfmathsetmacro{\Lfour}{\lfour * \mul}
    \pgfmathsetmacro{\Lone}{\lone * \mul}
    \pgfmathsetmacro{\Ltwo}{\ltwo * \mul}

    \coordinate (A) at (0, 0);
    \coordinate (B) at (-\Lthree, 0);
    \coordinate (C) at ({-\Lthree - \Lfour}, 0);
    \coordinate (D) at (0, -\Lone);
    \coordinate (E) at (0, {-\Lone - \Ltwo});
}

% --------------------------------------------------
% 3. BEAMS
% --------------------------------------------------
\newcommand{\beams}{
    % Main L-beam
    \draw[line cap=round, line join=round, line width=\beamwidth, \beamcolor] (C) -- (A) -- (E);

    % Connection joints (clean mechanical pins)
    \filldraw[fill=white, draw=black, thick] (A) circle (0.12) node[above right=0.15cm] {A};
    \filldraw[fill=white, draw=black, thick] (B) circle (0.12) node[above right=0.15cm] {B};
    \filldraw[fill=white, draw=black, thick] (C) circle (0.12) node[above left=0.1cm] {C};
    \filldraw[fill=white, draw=black, thick] (D) circle (0.12) node[above left=0.1cm] {D};

    % Diameter d marking (0.088 is exactly half of the 5pt beam width)
    \draw[<-] ($(C) + (1.2, 0.088)$) -- ++(0, 0.4);
    \draw[<-] ($(C) + (1.2, -0.088)$) -- ++(0, -0.4);
    \node at ($(C) + (1.7, 0.4)$) {$\varnothing d$};
}

% --------------------------------------------------
% 4. SUPPORTS, DAMPERS & SPRINGS
% --------------------------------------------------
\newcommand{\supports}{
    % Support at A (Triangle joint + Horizontal ceiling)
    \draw[thick] (0, 0.12) -- (-0.3, 0.5) -- (0.3, 0.5) -- cycle;
    \fill[pattern=north east lines] (-0.6, 0.5) rectangle (0.6, 0.7);
    \draw[thick] (-0.6, 0.5) -- (0.6, 0.5);
}

\newcommand{\springsdampers}{
    % ------------------------------------------------
    % Damper c1 at C (U-shape pot up, piston down)
    % ------------------------------------------------
    % Top rod and Piston
    \draw[thick] ($(C) + (0, -0.12)$) -- ($(C) + (0, -0.6)$);
    \draw[thick] ($(C) + (-0.15, -0.6)$) -- ($(C) + (0.15, -0.6)$);
    % Pot (U-shape opening upwards)
    \draw[thick] ($(C) + (-0.2, -0.3)$) -- ($(C) + (-0.2, -0.9)$) -- ($(C) + (0.2, -0.9)$) -- ($(C) + (0.2, -0.3)$);
    % Bottom rod to joint
    \draw[thick] ($(C) + (0, -0.9)$) -- ($(C) + (0, -1.4)$);
    % Ground joint (Csukló)
    \filldraw[fill=white, draw=black, thick] ($(C) + (0, -1.4)$) circle (0.08);
    \draw[thick] ($(C) + (0, -1.48)$) -- ($(C) + (-0.3, -1.9)$) -- ($(C) + (0.3, -1.9)$) -- cycle;
    \draw[thick] ($(C) + (-0.5, -1.9)$) -- ($(C) + (0.5, -1.9)$);
    \fill[pattern=north east lines] ($(C) + (-0.5, -2.1)$) rectangle ($(C) + (0.5, -1.9)$);
    \node at ($(C) + (0.6, -0.6)$) {$c_1$};

    % ------------------------------------------------
    % Spring k1 at B (Guided support block)
    % ------------------------------------------------
    \draw[thick, decoration={zigzag, pre length=3mm, post length=3mm, segment length=3mm, amplitude=2.5mm}, decorate] ($(B) + (0, -0.12)$) -- ($(B) + (0, -1.7)$);
    % Pin joint
    \filldraw[fill=white, draw=black, thick] ($(B) + (0, -1.7)$) circle (0.08);
    % Vertical guides (gray blocks)
    \fill[gray!40] ($(B) + (-0.3, -1.2)$) rectangle ($(B) + (-0.15, -2.2)$);
    \draw[thick] ($(B) + (-0.15, -1.2)$) -- ($(B) + (-0.15, -2.2)$);
    \fill[gray!40] ($(B) + (0.15, -1.2)$) rectangle ($(B) + (0.3, -2.2)$);
    \draw[thick] ($(B) + (0.15, -1.2)$) -- ($(B) + (0.15, -2.2)$);
    \node at ($(B) + (0.7, -1.0)$) {$k_1$};

    % ------------------------------------------------
    % Spring k2 at D (Triangle joint)
    % ------------------------------------------------
    \draw[thick, decoration={zigzag, pre length=3mm, post length=3mm, segment length=3mm, amplitude=2.5mm}, decorate] ($(D) + (-1.4, 0)$) -- ($(D) + (-0.12, 0)$);
    % Pin joint
    \filldraw[fill=white, draw=black, thick] ($(D) + (-1.4, 0)$) circle (0.08);
    % Triangle support to wall
    \draw[thick] ($(D) + (-1.48, 0)$) -- ($(D) + (-1.9, 0.3)$) -- ($(D) + (-1.9, -0.3)$) -- cycle;
    \draw[thick] ($(D) + (-1.9, -0.5)$) -- ($(D) + (-1.9, 0.5)$);
    \fill[pattern=north east lines] ($(D) + (-2.1, -0.5)$) rectangle ($(D) + (-1.9, 0.5)$);
    \node at ($(D) + (-0.9, 0.5)$) {$k_2$};
}

\newcommand{\spring}{
    % ------------------------------------------------
    % Torsional spring kt at A (Longer spiral, drawn in foreground)
    % ------------------------------------------------
    \begin{scope}[shift={(A)}]
        % Increased domain to 4.5*pi for more loops, adjusted radius scaling to fit
        \draw[domain=0:4.5*pi, variable=\t, samples=150, smooth, thick] plot ({\t r}: {0.1+\t*0.018});
        \node[right] at (0.6, -0.6) {$k_t$};
    \end{scope}
}

% --------------------------------------------------
% 5. DIMENSIONS & CONVENTIONS
% --------------------------------------------------
\newcommand{\dimensions}{
    % l4 & l3 (Top)
    \draw[|<->|] ($(C) + (0, 1.5)$) -- ($(B) + (0, 1.5)$) node[midway, above] {$l_4$};
    \draw[|<->|] ($(B) + (0, 1.5)$) -- ($(A) + (0, 1.5)$) node[midway, above] {$l_3$};
    \draw ($(C) + (0, 0.25)$) -- ($(C) + (0, 1.7)$);
    \draw ($(B) + (0, 0.25)$) -- ($(B) + (0, 1.7)$);
    \draw ($(A) + (0, 0.25)$) -- ($(A) + (0, 1.7)$);

    % l1 & l2 (Moved to the right side)
    \draw[|<->|] ($(A) + (1.2, 0)$) -- ($(D) + (1.2, 0)$) node[midway, right] {$l_1$};
    \draw[|<->|] ($(D) + (1.2, 0)$) -- ($(E) + (1.2, 0)$) node[midway, right] {$l_2$};
    \draw ($(A) + (0.12, 0)$) -- ($(A) + (1.4, 0)$);
    \draw ($(D) + (0.12, 0)$) -- ($(D) + (1.4, 0)$);
    \draw ($(E) + (0, 0)$) -- ($(E) + (1.4, 0)$);
}

\newcommand{\convention}{
    % Coordinate system (x, y)
    \draw[-Stealth, thick] (-8, -4) -- (-7, -4) node[below] {$x$};
    \draw[-Stealth, thick] (-8, -4) -- (-8, -3) node[left] {$y$};

    % Gravity g
    \draw[-Stealth, thick] (4, -1.0) -- (4, -2.0) node[midway, right] {$\mathbf{g}$};
}

% --------------------------------------------------
% 6. EXCITATIONS
% --------------------------------------------------
\newcommand{\excitations}{
    % Coordinate phi(t) 
    \draw[dashed, thick] (C) -- ++(-2.8, 0);
    % Solid line with a gap so it doesn't touch C (30 degree slope: 180 + 50 = 230)
    \draw[thick] ($(C) + (230:0.6)$) -- ($(C) + (210:2.8)$);
    % Angle arrow with a flat starting bar (|-Stealth), tracking from 180 to 210 degrees
    \draw[|-Stealth, green!60!black, thick] ($(C) + (-2.2, 0)$) arc (180:210:2.2) node[midway, right=0.08cm] {$\varphi(t)$};

    % Moment M(t) at A
    \begin{scope}[shift={(A)}]
        \draw[-Stealth, thick, brown!80!black] (0.7, -1.2) arc (-70:80:1.3) node[midway, right] {$M(t)$};
    \end{scope}
}

% --------------------------------------------------
% 7. MAIN ASSEMBLY MACRO

\newcommand{\structurediagram}{
    \centering
    \begin{tikzpicture}
        \coords
        \supports
        \springsdampers
        \excitations
        \beams
        \spring
        \dimensions
        \convention
    \end{tikzpicture}
}
